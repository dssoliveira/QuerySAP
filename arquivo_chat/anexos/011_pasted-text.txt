        0

NF_BASE AS (

    /* =========================================================
       BASE FISCAL OFICIAL DE FATURAMENTO

       Regra:
       - Valor oficial da NF-e: OINV."NfeValue"
       - Uma linha por NF para evitar duplicidade
       - Centro de custo vindo da linha da NF
         ou, se necessário, da linha do pedido de venda
       - Exclui canceladas
       - Exclui NF de entrega baseada em entrega/remessa
       - Exclui linhas somente imposto
       - Exclui utilização 37: remessa comodato
    ========================================================= */

    SELECT
        X."DocEntry",
        X."DocDate",
        X."NfeValue",
        X."OcrCode"

    FROM (

        SELECT
            T0."DocEntry",
            T0."DocDate",
            T0."NfeValue",

            IFNULL(T1."OcrCode", R1."OcrCode") AS "OcrCode",

            ROW_NUMBER() OVER (
                PARTITION BY T0."DocEntry"
                ORDER BY
                    CASE
                        WHEN T1."BaseType" = 17 THEN 0
                        ELSE 1
                    END,
                    T1."LineNum"
            ) AS "LinhaControle"

        FROM OINV T0

        INNER JOIN INV1 T1
            ON T1."DocEntry" = T0."DocEntry"

        LEFT JOIN RDR1 R1
            ON T1."BaseType" = 17
            AND T1."BaseEntry" = R1."DocEntry"
            AND T1."BaseLine" = R1."LineNum"

        LEFT JOIN OUSG U
            ON U."ID" = T1."Usage"

        WHERE
            T0."CANCELED" = 'N'
            AND YEAR(T0."DocDate") = YEAR(CURRENT_DATE)

            AND IFNULL(T1."OcrCode", R1."OcrCode") IN (
                'GERADOR',
                'NOBREAK',
                'LOCACAO',
                'EXE_ELET',
                'FOTOV',
                'REV_DIRE',
                'SEMINOVO'
            )

            AND IFNULL(T1."TaxOnly", 'N') <> 'Y'
            AND IFNULL(U."TaxOnly", 'N') <> 'Y'
            AND IFNULL(T1."Usage", -1) <> 37

            AND NOT EXISTS (
                SELECT
                    1
                FROM INV1 E
                WHERE
                    E."DocEntry" = T0."DocEntry"
                    AND E."BaseType" = 15
            )

    ) X

    WHERE
        X."LinhaControle" = 1
),

A AS (

    /* =========================================================
       FATURAMENTO REAL POR CENTRO DE CUSTO
       Valor utilizado: OINV."NfeValue"
    ========================================================= */

    SELECT
        NF."OcrCode",

        SUM(
            CASE WHEN MONTH(NF."DocDate") = 1
            THEN NF."NfeValue" ELSE 0 END
        ) AS "Jan",

        SUM(
            CASE WHEN MONTH(NF."DocDate") = 2
            THEN NF."NfeValue" ELSE 0 END
        ) AS "Fev",

        SUM(
            CASE WHEN MONTH(NF."DocDate") = 3
            THEN NF."NfeValue" ELSE 0 END
        ) AS "Mar",

        SUM(
            CASE WHEN MONTH(NF."DocDate") = 4
            THEN NF."NfeValue" ELSE 0 END
        ) AS "Abr",

        SUM(
            CASE WHEN MONTH(NF."DocDate") = 5
            THEN NF."NfeValue" ELSE 0 END
        ) AS "Mai",

        SUM(
            CASE WHEN MONTH(NF."DocDate") = 6
            THEN NF."NfeValue" ELSE 0 END
        ) AS "Jun",

        SUM(
            CASE WHEN MONTH(NF."DocDate") = 7
            THEN NF."NfeValue" ELSE 0 END
        ) AS "Jul",

        SUM(
            CASE WHEN MONTH(NF."DocDate") = 8
            THEN NF."NfeValue" ELSE 0 END
        ) AS "Ago",

        SUM(
            CASE WHEN MONTH(NF."DocDate") = 9
            THEN NF."NfeValue" ELSE 0 END
        ) AS "Set",

        SUM(
            CASE WHEN MONTH(NF."DocDate") = 10
            THEN NF."NfeValue" ELSE 0 END
        ) AS "Out",

        SUM(
            CASE WHEN MONTH(NF."DocDate") = 11
            THEN NF."NfeValue" ELSE 0 END
        ) AS "Nov",

        SUM(
            CASE WHEN MONTH(NF."DocDate") = 12
            THEN NF."NfeValue" ELSE 0 END
        ) AS "Dez",

        SUM(NF."NfeValue") AS "AcumuladoSAP"

    FROM NF_BASE NF

    GROUP BY
        NF."OcrCode"
),

R AS (

    /* =========================================================
       META
    ========================================================= */

    SELECT
        M."OrdemDepto" * 2 - 1 AS "Ordem",
        M."Departamento" AS "Departamento",
        M."OcrCode" AS "OcrCode",
        'Meta' AS "Indicador",

        M."MetaJan" AS "Jan",
        M."MetaFev" AS "Fev",
        M."MetaMar" AS "Mar",
        M."MetaAbr" AS "Abr",
        M."MetaMai" AS "Mai",
        M."MetaJun" AS "Jun",
        M."MetaJul" AS "Jul",
        M."MetaAgo" AS "Ago",
        M."MetaSet" AS "Set",
        M."MetaOut" AS "Out",
        M."MetaNov" AS "Nov",
        M."MetaDez" AS "Dez",

        M."MetaAno" AS "Acumulado",

        NULL AS "%Alcancado"

    FROM M

    UNION ALL

    /* =========================================================
       REALIZADO
    ========================================================= */

    SELECT
        M."OrdemDepto" * 2 AS "Ordem",
        '' AS "Departamento",
        M."OcrCode" AS "OcrCode",
        'Realizado' AS "Indicador",

        IFNULL(A."Jan", 0) + M."RetroJan" AS "Jan",
        IFNULL(A."Fev", 0) + M."RetroFev" AS "Fev",
        IFNULL(A."Mar", 0) AS "Mar",
        IFNULL(A."Abr", 0) AS "Abr",
        IFNULL(A."Mai", 0) AS "Mai",
        IFNULL(A."Jun", 0) AS "Jun",
        IFNULL(A."Jul", 0) AS "Jul",
        IFNULL(A."Ago", 0) AS "Ago",
        IFNULL(A."Set", 0) AS "Set",
        IFNULL(A."Out", 0) AS "Out",
        IFNULL(A."Nov", 0) AS "Nov",
        IFNULL(A."Dez", 0) AS "Dez",

        IFNULL(A."AcumuladoSAP", 0)
            + M."RetroJan"
            + M."RetroFev" AS "Acumulado",

        CASE
            WHEN M."MetaAno" = 0 THEN 0
            ELSE ROUND(
                (
                    IFNULL(A."AcumuladoSAP", 0)
                    + M."RetroJan"
                    + M."RetroFev"
                ) / M."MetaAno" * 100,
                2
            )
        END AS "%Alcancado"

    FROM M

    LEFT JOIN A
        ON A."OcrCode" = M."OcrCode"
),

S AS (

    /* =========================================================
       SOMATORIA EXE / FV / RVD / SN
    ========================================================= */

    SELECT
        R."Indicador",

        SUM(R."Jan") AS "Jan",
        SUM(R."Fev") AS "Fev",
        SUM(R."Mar") AS "Mar",
        SUM(R."Abr") AS "Abr",
        SUM(R."Mai") AS "Mai",
        SUM(R."Jun") AS "Jun",
        SUM(R."Jul") AS "Jul",
        SUM(R."Ago") AS "Ago",
        SUM(R."Set") AS "Set",
        SUM(R."Out") AS "Out",
        SUM(R."Nov") AS "Nov",
        SUM(R."Dez") AS "Dez",

        SUM(R."Acumulado") AS "Acumulado"

    FROM R

    WHERE
        R."OcrCode" IN (
            'EXE_ELET',
            'FOTOV',
            'REV_DIRE',
            'SEMINOVO'
        )

    GROUP BY
        R."Indicador"
),

T AS (

    /* =========================================================
       SUBTOTAL GERAL
    ========================================================= */

    SELECT
        R."Indicador",

        SUM(R."Jan") AS "Jan",
        SUM(R."Fev") AS "Fev",

        SUM(R."Abr") AS "Abr",
        SUM(R."Mai") AS "Mai",
        SUM(R."Jun") AS "Jun",
        SUM(R."Jul") AS "Jul",
        SUM(R."Ago") AS "Ago",
        SUM(R."Set") AS "Set",
        SUM(R."Out") AS "Out",
        SUM(R."Nov") AS "Nov",
        SUM(R."Dez") AS "Dez",

        SUM(R."Acumulado") AS "Acumulado"

    FROM R

    GROUP BY
        R."Indicador"
)

SELECT
    X."Departamento",
    X."Indicador",
    X."Jan",
    X."Fev",
    X."Mar",
    X."Abr",
    X."Mai",
    X."Jun",
    X."Jul",
    X."Ago",
    X."Set",
    X."Out",
    X."Nov",
    X."Dez",
    X."Acumulado",
    X."%Alcancado"

FROM (

    /* =========================================================
       DEPARTAMENTOS
    ========================================================= */

    SELECT
        R."Ordem",
        R."Departamento",
        R."Indicador",
        R."Jan",
        R."Fev",
        R."Mar",

        R."Jun",

        R."Ago",
        R."Set",

        R."Nov",
        R."Dez",
        R."Acumulado",
        R."%Alcancado"

    FROM R

    UNION ALL

    /* =========================================================
       SOMATORIA EXE/FV/RVD/SN - META
    ========================================================= */

    SELECT
        997 AS "Ordem",
        'Somatoria EXE/FV/RVD/SN' AS "Departamento",
        'Meta' AS "Indicador",

        SM."Jan",
        SM."Fev",
        SM."Mar",
        SM."Abr",
        SM."Mai",
        SM."Jun",
        SM."Jul",
        SM."Ago",
        SM."Set",
        SM."Out",
        SM."Nov",
        SM."Dez",
        SM."Acumulado",

        NULL AS "%Alcancado"

    FROM S SM

    WHERE
        SM."Indicador" = 'Meta'

    UNION ALL

    /* =========================================================
       SOMATORIA EXE/FV/RVD/SN - REALIZADO
    ========================================================= */

    SELECT
        998 AS "Ordem",
        '' AS "Departamento",
        'Realizado' AS "Indicador",

        SR."Jan",
        SR."Fev",
        SR."Mar",
        SR."Abr",
        SR."Mai",
        SR."Jun",
        SR."Jul",
        SR."Ago",
        SR."Set",
        SR."Out",
        SR."Nov",
        SR."Dez",
        SR."Acumulado",

        CASE
            WHEN IFNULL(SM."Acumulado", 0) = 0 THEN 0
            ELSE ROUND(
                SR."Acumulado" / SM."Acumulado" * 100,
                2
            )
        END AS "%Alcancado"

    FROM S SR

    LEFT JOIN S SM
        ON SM."Indicador" = 'Meta'

    WHERE
        SR."Indicador" = 'Realizado'

    UNION ALL

    /* =========================================================
       SUBTOTAL GERAL - META
    ========================================================= */

    SELECT
        999 AS "Ordem",
        'Subtotal' AS "Departamento",
        'Meta' AS "Indicador",

        TM."Jan",
        TM."Fev",
        TM."Mar",

        TM."Ago",

        TM."Out",
        TM."Nov",
        TM."Dez",
        TM."Acumulado",

        NULL AS "%Alcancado"

    FROM T TM

    WHERE
        TM."Indicador" = 'Meta'

    UNION ALL

    /* =========================================================
       SUBTOTAL GERAL - REALIZADO
    ========================================================= */

    SELECT
        1000 AS "Ordem",
        '' AS "Departamento",
        'Realizado' AS "Indicador",

        TR."Jan",
        TR."Fev",
        TR."Mar",
        TR."Abr",
        TR."Mai",
        TR."Jun",
        TR."Jul",
        TR."Ago",
        TR."Set",
        TR."Out",
        TR."Nov",
        TR."Dez",
        TR."Acumulado",

        CASE
            WHEN IFNULL(TM."Acumulado", 0) = 0 THEN 0
            ELSE ROUND(
                TR."Acumulado" / TM."Acumulado" * 100,
                2
            )
        END AS "%Alcancado"

    FROM T TR

    LEFT JOIN T TM
        ON TM."Indicador" = 'Meta'

        TR."Indicador" = 'Realizado'
